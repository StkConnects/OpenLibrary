# OpenLibrary — OpenRemote Hovercraft firmware

cd ~/Projects/OpenLibrary/projects/OpenRemote/profiles/Hovercraft/firmware/OpenRemoteESP32

# Compile
arduino-cli compile --fqbn esp32:esp32:esp32 .

# Upload (connect the ESP32 and verify its port first)
arduino-cli upload -p /dev/ttyUSB0 --fqbn esp32:esp32:esp32 .

# Serial monitor
arduino-cli monitor --port /dev/ttyUSB0 --config baudrate=115200
